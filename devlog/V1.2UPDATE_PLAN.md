# V1.1 UPDATE PLAN: 카카오톡 알림 기능 추가

## 개요

기존 인앱 알림(컴플레인/클레임 리뷰 감지, 자동화 보류/승인 대기)에 더해 **카카오톡 메시지 알림 채널**을 추가한다. 병원 관리자가 카카오 로그인을 통해 계정을 연동하면, 주요 알림 발생 시 카카오톡으로 실시간 알림을 받을 수 있다.

### 현재 상태

| 항목 | 현재 | 한계 |
|------|------|------|
| 알림 채널 | 인앱 알림만 지원 | 앱 미접속 시 알림 확인 불가 |
| 알림 전달 | DB 저장 → 프론트엔드 조회 | 실시간 푸시 없음, 폴링 방식 |
| 외부 연동 | 없음 | 카카오/문자 등 외부 알림 채널 부재 |

### 개편 목표

```
[반감 리뷰 감지 / 자동화 이벤트 발생]
    │
    ▼
[alert_service: 인앱 알림 생성 (기존)]
    │
    ├─ 카카오톡 알림 활성화됨?
    │       │
    │       ├── YES → kakao_service: 카카오톡 메시지 발송
    │       │         ├── 나에게 보내기 (기본)
    │       │         └── 친구에게 보내기 (선택)
    │       │
    │       └── NO → 인앱 알림만 유지
    │
    ▼
[관리자: 카카오톡 + 인앱 알림 동시 수신]
```

### 1-1. 카카오 개발자 설정 사전 준비

6. (선택) `friends` 권한 활성화 — 친구에게 메시지 발송 시 필요

### 1-4. 카카오 OAuth 라우터

**파일**: `backend/routers/kakao.py` (신규)

카카오 로그인 플로우를 처리하는 라우터를 추가한다.

```python
"""
카카오 OAuth 및 알림 API 라우터
"""
from fastapi import APIRouter, Depends, HTTPException
from fastapi.responses import RedirectResponse
from sqlalchemy.orm import Session

from config import settings
from database import get_db
from services.kakao_service import (
    exchange_code_for_token,
    get_kakao_profile,
    refresh_kakao_token,
    disconnect_kakao,
)

router = APIRouter(prefix="/api/kakao", tags=["kakao"])


@router.get("/authorize")
def kakao_authorize():
    """카카오 로그인 페이지로 리다이렉트"""
    scope = "talk_message,friends"
    authorize_url = (
        f"https://kauth.kakao.com/oauth/authorize"
        f"?response_type=code"
        f"&client_id={settings.KAKAO_CLIENT_ID}"
        f"&redirect_uri={settings.KAKAO_REDIRECT_URI}"
        f"&scope={scope}"
    )
    return RedirectResponse(url=authorize_url)


@router.get("/redirect")
def kakao_redirect(code: str, db: Session = Depends(get_db)):
    """
    카카오 OAuth 콜백 — 인가 코드로 토큰 발급 후 Hospital에 저장
    """
    token_data = exchange_code_for_token(code)

    # 카카오 프로필로 사용자 식별
    profile = get_kakao_profile(token_data["access_token"])

    # Hospital 업데이트 (MVP: 단일 병원)
    from models import Hospital
    hospital = db.query(Hospital).first()
    if not hospital:
        raise HTTPException(status_code=404, detail="병원 정보가 없습니다. 먼저 병원을 등록하세요.")

    hospital.kakao_access_token = token_data["access_token"]
    hospital.kakao_refresh_token = token_data.get("refresh_token")
    hospital.kakao_token_expires_at = token_data.get("expires_at")
    hospital.kakao_user_id = str(profile["id"])
    hospital.kakao_notify_enabled = True
    db.commit()

    # 프론트엔드 설정 페이지로 리다이렉트
    return RedirectResponse(url="/settings?kakao=connected")


@router.post("/disconnect")
def kakao_disconnect(db: Session = Depends(get_db)):
    """카카오 연동 해제"""
    from models import Hospital
    hospital = db.query(Hospital).first()
    if not hospital or not hospital.kakao_access_token:
        raise HTTPException(status_code=400, detail="카카오 연동 상태가 아닙니다.")

    disconnect_kakao(hospital.kakao_access_token)

    hospital.kakao_access_token = None
    hospital.kakao_refresh_token = None
    hospital.kakao_token_expires_at = None
    hospital.kakao_user_id = None
    hospital.kakao_notify_enabled = False
    db.commit()

    return {"message": "카카오 연동이 해제되었습니다."}


@router.post("/test-message")
def send_test_message(db: Session = Depends(get_db)):
    """카카오톡 테스트 메시지 발송"""
    from models import Hospital
    from services.kakao_service import send_kakao_alert

    hospital = db.query(Hospital).first()
    if not hospital or not hospital.kakao_access_token:
        raise HTTPException(status_code=400, detail="카카오 연동이 필요합니다.")

    success = send_kakao_alert(
        db=db,
        hospital=hospital,
        title="🔔 MediReview 테스트 알림",
        message="카카오톡 알림이 정상적으로 연동되었습니다.",
        link_path="/settings",
    )

    if not success:
        raise HTTPException(status_code=500, detail="메시지 발송 실패")

    return {"message": "테스트 메시지가 발송되었습니다."}
```

## Phase 2: 카카오톡 메시지 전송 서비스

### 2-1. 카카오 서비스 모듈

**파일**: `backend/services/kakao_service.py` (신규)

카카오 API 호출을 담당하는 서비스 모듈을 추가한다. `tmp/login_and_message_example/api.py`의 인증/메시지 발송 패턴을 참고하여 FastAPI + SQLAlchemy 환경에 맞게 구현한다.

```python
"""
카카오톡 알림 서비스
- OAuth 토큰 교환/갱신
- 나에게 메시지 발송 (기본 텍스트 템플릿)
"""
import json
import logging
from datetime import datetime, timedelta

import requests
from sqlalchemy.orm import Session

from config import settings

logger = logging.getLogger(__name__)

KAUTH_HOST = "https://kauth.kakao.com"
KAPI_HOST = "https://kapi.kakao.com"


def exchange_code_for_token(code: str) -> dict:
    """
    인가 코드 → 액세스/리프레시 토큰 교환

    카카오 OAuth 토큰 발급 API를 호출한다.
    참고: tmp/login_and_message_example/api.py - redirect_page()
    """
    data = {
        "grant_type": "authorization_code",
        "client_id": settings.KAKAO_CLIENT_ID,
        "redirect_uri": settings.KAKAO_REDIRECT_URI,
        "client_secret": settings.KAKAO_CLIENT_SECRET,
        "code": code,
    }
    resp = requests.post(f"{KAUTH_HOST}/oauth/token", data=data, timeout=10)
    resp.raise_for_status()
    token_data = resp.json()

    # 만료 시각 계산
    expires_in = token_data.get("expires_in", 7200)
    token_data["expires_at"] = datetime.utcnow() + timedelta(seconds=expires_in)

    return token_data


def refresh_kakao_token(hospital) -> bool:
    """
    리프레시 토큰으로 액세스 토큰 갱신

    Returns: 갱신 성공 여부
    """
    if not hospital.kakao_refresh_token:
        return False

    data = {
        "grant_type": "refresh_token",
        "client_id": settings.KAKAO_CLIENT_ID,
        "client_secret": settings.KAKAO_CLIENT_SECRET,
        "refresh_token": hospital.kakao_refresh_token,
    }
    resp = requests.post(f"{KAUTH_HOST}/oauth/token", data=data, timeout=10)

    if resp.status_code != 200:
        logger.error(f"카카오 토큰 갱신 실패: {resp.text}")
        return False

    token_data = resp.json()
    hospital.kakao_access_token = token_data["access_token"]

    if "refresh_token" in token_data:
        hospital.kakao_refresh_token = token_data["refresh_token"]

    expires_in = token_data.get("expires_in", 7200)
    hospital.kakao_token_expires_at = datetime.utcnow() + timedelta(seconds=expires_in)

    return True


def get_kakao_profile(access_token: str) -> dict:
    """카카오 사용자 프로필 조회"""
    headers = {"Authorization": f"Bearer {access_token}"}
    resp = requests.get(f"{KAPI_HOST}/v2/user/me", headers=headers, timeout=10)
    resp.raise_for_status()
    return resp.json()


def disconnect_kakao(access_token: str) -> bool:
    """카카오 연결 끊기"""
    headers = {"Authorization": f"Bearer {access_token}"}
    resp = requests.post(f"{KAPI_HOST}/v1/user/unlink", headers=headers, timeout=10)
    return resp.status_code == 200


def _ensure_valid_token(db: Session, hospital) -> bool:
    """토큰 만료 확인 후 필요 시 갱신"""
    if not hospital.kakao_access_token:
        return False

    if hospital.kakao_token_expires_at and hospital.kakao_token_expires_at < datetime.utcnow():
        logger.info("카카오 액세스 토큰 만료 — 갱신 시도")
        success = refresh_kakao_token(hospital)
        if success:
            db.commit()
        return success

    return True


def send_kakao_alert(
    db: Session,
    hospital,
    title: str,
    message: str,
    link_path: str = "/",
) -> bool:
    """
    카카오톡 '나에게 보내기'로 알림 메시지 발송

    참고: tmp/login_and_message_example/api.py - message()
    나에게 보내기 API: POST /v2/api/talk/memo/default/send

    Args:
        db: DB 세션 (토큰 갱신 시 커밋용)
        hospital: Hospital 모델 객체
        title: 메시지 제목
        message: 메시지 본문
        link_path: 클릭 시 이동할 앱 내 경로

    Returns: 발송 성공 여부
    """
    if not hospital.kakao_notify_enabled:
        return False

    if not _ensure_valid_token(db, hospital):
        logger.warning("카카오 토큰이 유효하지 않아 메시지 발송 불가")
        return False

    # 카카오 기본 텍스트 메시지 템플릿
    # 참고: tmp/login_and_message_example/api.py의 message_template 구조
    frontend_url = "http://localhost:5173"
    template_object = {
        "object_type": "text",
        "text": f"[{title}]\n\n{message}",
        "link": {
            "web_url": f"{frontend_url}{link_path}",
            "mobile_web_url": f"{frontend_url}{link_path}",
        },
        "button_title": "MediReview에서 확인",
    }

    headers = {"Authorization": f"Bearer {hospital.kakao_access_token}"}
    data = {"template_object": json.dumps(template_object)}

    try:
        resp = requests.post(
            f"{KAPI_HOST}/v2/api/talk/memo/default/send",
            headers=headers,
            data=data,
            timeout=10,
        )

        if resp.status_code == 200:
            logger.info(f"카카오톡 알림 발송 성공: {title}")
            return True
        else:
            logger.error(f"카카오톡 알림 발송 실패 [{resp.status_code}]: {resp.text}")
            return False
    except requests.RequestException as e:
        logger.error(f"카카오톡 알림 발송 중 오류: {e}")
        return False
```

### 2-2. 메시지 템플릿 정의

알림 유형별로 발송되는 메시지 형식을 정의한다:

| 알림 유형 | 제목 | 본문 예시 |
|-----------|------|-----------|
| `complaint` | 🟠 컴플레인 리뷰 감지 | `"{author}"님이 {platform}에 {rating}점 리뷰를 남겼습니다.\n"{content 앞 50자}..."` |
| `claim` | 🔴 클레임 리뷰 감지 | `"{author}"님이 {platform}에 {rating}점 리뷰를 남겼습니다.\n"{content 앞 50자}..."` |
| `automation_held` | 🚫 자동 대응 보류 | `키워드 "{keyword}"가 감지되어 자동 대응이 보류되었습니다.` |
| `approval_needed` | ⏳ 승인 대기 | `AI 댓글이 생성되었습니다. 승인이 필요합니다.` |

**파일**: `backend/services/kakao_service.py` 에 템플릿 함수 추가:

```python
def build_alert_message(alert_type: str, review=None, reason: str = "") -> tuple[str, str, str]:
    """
    알림 유형에 따른 카카오톡 메시지 생성

    Returns: (title, message, link_path)
    """
    content_preview = ""
    if review and review.content:
        content_preview = review.content[:50] + ("..." if len(review.content) > 50 else "")

    templates = {
        "complaint": (
            "🟠 컴플레인 리뷰 감지",
            f'"{review.author or "익명"}"님이 {review.platform}에 {review.rating}점 리뷰를 남겼습니다.\n"{content_preview}"',
            f"/reviews/{review.id}" if review else "/alerts",
        ),
        "claim": (
            "🔴 클레임 리뷰 감지",
            f'"{review.author or "익명"}"님이 {review.platform}에 {review.rating}점 리뷰를 남겼습니다.\n"{content_preview}"',
            f"/reviews/{review.id}" if review else "/alerts",
        ),
        "automation_held": (
            "🚫 자동 대응 보류",
            f"키워드가 감지되어 자동 대응이 보류되었습니다.\n사유: {reason}",
            f"/reviews/{review.id}" if review else "/alerts",
        ),
        "approval_needed": (
            "⏳ AI 댓글 승인 대기",
            "AI 댓글이 생성되었습니다. 승인이 필요합니다.",
            f"/reviews/{review.id}" if review else "/alerts",
        ),
    }

    return templates.get(alert_type, ("🔔 MediReview 알림", "새로운 알림이 있습니다.", "/alerts"))
```

## Phase 3: 알림 서비스 통합

### 3-1. alert_service.py 수정

**파일**: `backend/services/alert_service.py`

기존 `check_and_create_alert()` 함수에 카카오톡 발송 로직을 추가한다:

```python
def check_and_create_alert(db: Session, review: Review) -> Alert | None:
    """
    리뷰 유형이 컴플레인 또는 클레임이면 알림 생성
    + 카카오톡 알림 활성화 시 카카오톡 메시지 발송
    """
    alert_type = _ALERT_TYPE_MAP.get(review.review_type)
    if not alert_type:
        return None

    # 중복 알림 체크
    existing = (
        db.query(Alert)
        .filter(Alert.review_id == review.id)
        .first()
    )
    if existing:
        return None

    alert = Alert(
        hospital_id=review.hospital_id,
        review_id=review.id,
        type=alert_type,
    )
    db.add(alert)
    db.commit()
    db.refresh(alert)

    # ── 카카오톡 알림 발송 (V1.1 추가) ──
    _send_kakao_notification(db, review.hospital_id, alert_type, review)

    return alert


def _send_kakao_notification(db: Session, hospital_id: int, alert_type: str, review: Review):
    """카카오톡 알림 발송 (실패해도 인앱 알림에 영향 없음)"""
    try:
        from models import Hospital
        from services.kakao_service import send_kakao_alert, build_alert_message

        hospital = db.query(Hospital).filter(Hospital.id == hospital_id).first()
        if not hospital or not hospital.kakao_notify_enabled:
            return

        title, message, link_path = build_alert_message(alert_type, review)
        send_kakao_alert(db, hospital, title, message, link_path)
    except Exception as e:
        logger.warning(f"카카오톡 알림 발송 실패 (인앱 알림은 정상): {e}")
```

### 5-2. DB 마이그레이션

**파일**: `backend/migrations/v1_1_update.py` (신규)

```python
"""
V1.1 마이그레이션: Hospital 테이블에 카카오 연동 컬럼 추가
"""
import sqlite3

def migrate(db_path: str = "medireview.db"):
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()

    columns_to_add = [
        ("kakao_access_token", "TEXT"),
        ("kakao_refresh_token", "TEXT"),
        ("kakao_token_expires_at", "DATETIME"),
        ("kakao_notify_enabled", "BOOLEAN DEFAULT 0"),
        ("kakao_user_id", "VARCHAR(50)"),
    ]

    for col_name, col_type in columns_to_add:
        try:
            cursor.execute(f"ALTER TABLE hospitals ADD COLUMN {col_name} {col_type}")
        except sqlite3.OperationalError:
            pass  # 이미 존재

    conn.commit()
    conn.close()

if __name__ == "__main__":
    migrate()
```

## 흐름 정리

### 카카오 연동 흐름

```
[관리자: 설정 페이지 → "카카오로 로그인하여 연동하기" 클릭]
    │
    ▼
[Backend: GET /api/kakao/authorize]
    │
    ▼
[카카오 로그인 페이지 (kauth.kakao.com)]
    │ (사용자 동의 후 인가 코드 발급)
    ▼
[Backend: GET /api/kakao/redirect?code=xxx]
    │
    ├─ POST kauth.kakao.com/oauth/token → 액세스/리프레시 토큰 발급
    ├─ GET kapi.kakao.com/v2/user/me → 사용자 ID 확인
    └─ Hospital 테이블에 토큰 저장 + kakao_notify_enabled = True
    │
    ▼
[프론트엔드: /settings?kakao=connected → "연동 완료" 표시]
```

### 알림 발송 흐름

```
[리뷰 수집 → 반감 리뷰 감지]
    │
    ▼
[alert_service.check_and_create_alert()]
    │
    ├─ 인앱 Alert 생성 (기존)
    │
    └─ hospital.kakao_notify_enabled == True ?
            │
            ├── YES
            │     ├─ 토큰 만료? → refresh_kakao_token() → 갱신
            │     ├─ build_alert_message() → 유형별 메시지 생성
            │     └─ send_kakao_alert() → 카카오톡 나에게 보내기
            │
            └── NO → 종료 (인앱 알림만)
```
